    function navigateTo(pageId) {
      activePage = pageId;
      document.querySelectorAll('.view-section').forEach(sec => sec.classList.add('hidden'));
      const target = document.getElementById(`view-${pageId}`);
      if (target) target.classList.remove('hidden');

      // Update Header Navigation Active State
      document.querySelectorAll('.nav-btn').forEach(btn => {
        if (btn.dataset.nav === pageId) {
          btn.classList.add('bg-white', 'text-brand-dark', 'shadow-sm');
          btn.classList.remove('text-brand-muted');
        } else {
          btn.classList.remove('bg-white', 'text-brand-dark', 'shadow-sm');
          btn.classList.add('text-brand-muted');
        }
      });

      // Update Sidebar Items Active State
      document.querySelectorAll('.sidebar-item').forEach(item => {
        if (item.dataset.page === pageId) {
          item.classList.add('bg-brand-50', 'text-brand-dark', 'font-bold');
          item.classList.remove('text-brand-muted');
        } else {
          item.classList.remove('bg-brand-50', 'text-brand-dark', 'font-bold');
          item.classList.add('text-brand-muted');
        }
      });

      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function setTextSize(size) {
      currentTextSize = size;
      const html = document.documentElement;
      html.classList.remove('text-sm-mode', 'text-md-mode', 'text-lg-mode');
      html.classList.add(`text-${size}-mode`);

      ['sm', 'md', 'lg'].forEach(s => {
        const btnHeader = document.getElementById(`ts-${s}`);
        const btnSet = document.getElementById(`set-ts-${s}`);
        
        if (s === size) {
          if (btnHeader) btnHeader.className = 'px-2 py-1 text-xs font-bold rounded bg-white shadow-sm text-brand-500 transition';
          if (btnSet) btnSet.className = 'px-4 py-2 text-sm font-bold rounded-xl bg-white text-brand-500 shadow-sm transition';
        } else {
          if (btnHeader) btnHeader.className = 'px-2 py-1 text-xs font-bold rounded hover:bg-white transition text-brand-muted';
          if (btnSet) btnSet.className = 'px-4 py-2 text-sm font-bold rounded-xl hover:bg-white transition text-brand-muted';
        }
      });
    }

    function filterCategory(cat) {
      document.querySelectorAll('.cat-chip').forEach(btn => {
        if (btn.dataset.cat === cat) {
          btn.className = 'cat-chip bg-brand-500 text-white px-4 py-2 rounded-full text-xs font-bold shrink-0 transition';
        } else {
          btn.className = 'cat-chip bg-white text-brand-dark border border-brand-500/10 px-4 py-2 rounded-full text-xs font-bold shrink-0 hover:bg-brand-50 transition';
        }
      });

      if (cat === 'all') {
        renderCourseCatalog(coursesData);
      } else {
        const filtered = coursesData.filter(c => c.category === cat);
        renderCourseCatalog(filtered);
      }
    }

    function filterCourses() {
      const query = document.getElementById('course-search-input').value.toLowerCase();
      const filtered = coursesData.filter(c => c.title.toLowerCase().includes(query) || c.description.toLowerCase().includes(query));
      renderCourseCatalog(filtered);
    }

    function nextOnboardingStep(step) {
      document.getElementById('ob-step-1-content').classList.add('hidden');
      document.getElementById('ob-step-2-content').classList.add('hidden');
      document.getElementById('ob-step-3-content').classList.add('hidden');

      document.getElementById('ob-step-label').innerText = `Шаг ${step} из 3`;

      [1, 2, 3].forEach(s => {
        const bar = document.getElementById(`ob-bar-${s}`);
        if (s <= step) {
          bar.className = 'w-6 h-2 rounded-full bg-brand-500';
        } else {
          bar.className = 'w-6 h-2 rounded-full bg-gray-200';
        }
      });

      if (step === 2) {
        renderInterestsChips();
      } else if (step === 3) {
        renderGoalsList();
      }
    }
